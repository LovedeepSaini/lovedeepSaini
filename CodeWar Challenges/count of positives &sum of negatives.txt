function countPositivesSumNegatives(input) {
  let arr=[];
  if(input===null || input.length<1){
    return [];
  }
  let count=0;
  let counter=0
  for(let i=0;i<input.length;i++){
    if(input[i]>0){
      count++
      
    }else{
      counter+=input[i]
      
    }
  }
  arr.push(count)
  arr.push(counter)
  return arr
  // your code here
}
